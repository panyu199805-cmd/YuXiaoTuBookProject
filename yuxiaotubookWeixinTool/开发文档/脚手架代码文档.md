## 六、核心代码模板

### 6.1 HTTP 客户端 `src/api/client.ts`

与 Web `frontend/src/api/client.ts` 同构，改用 `uni.request`：

```typescript
import { API_BASE } from '@/config/env'
import { getToken, clearToken } from '@/utils/storage'

export interface ApiResponse<T = unknown> {
  code: number
  msg: string
  data: T
}

interface RequestOptions {
  url: string
  method?: 'GET' | 'POST' | 'PUT' | 'DELETE'
  data?: Record<string, unknown>
  auth?: boolean
}

export function request<T>(opts: RequestOptions): Promise<T> {
  const { url, method = 'GET', data, auth = true } = opts
  const header: Record<string, string> = {
    'Content-Type': 'application/json',
  }
  if (auth) {
    const token = getToken()
    if (token) header.Authorization = `Bearer ${token}`
  }

  return new Promise((resolve, reject) => {
    uni.request({
      url: `${API_BASE}${url}`,
      method,
      data,
      header,
      success: (res) => {
        const body = res.data as ApiResponse<T>
        if (res.statusCode === 401 || body.code === 401) {
          clearToken()
          const pages = getCurrentPages()
          const current = pages[pages.length - 1]
          const route = current ? `/${current.route}` : '/pages/index/index'
          uni.navigateTo({ url: `/pages/auth/login?redirect=${encodeURIComponent(route)}` })
          reject(new Error('未登录'))
          return
        }
        if (body.code !== 200) {
          uni.showToast({ title: body.msg || '请求失败', icon: 'none' })
          reject(new Error(body.msg))
          return
        }
        resolve(body.data)
      },
      fail: (err) => {
        uni.showToast({ title: '网络异常', icon: 'none' })
        reject(err)
      },
    })
  })
}

export const get = <T>(url: string, auth = true) =>
  request<T>({ url, method: 'GET', auth })

export const post = <T>(url: string, data?: Record<string, unknown>, auth = true) =>
  request<T>({ url, method: 'POST', data, auth })
```

### 6.3 媒体 URL `src/utils/media.ts`

```typescript
import { MEDIA_ORIGIN } from '@/config/env'

export function resolveMediaUrl(path?: string | null): string {
  if (!path) return ''
  if (path.startsWith('http://') || path.startsWith('https://')) return path
  const base = MEDIA_ORIGIN.replace(/\/$/, '')
  const p = path.startsWith('/') ? path : `/${path}`
  return `${base}${p}`
}
```

### 6.5 Site Store `src/stores/site.ts`

对齐 Web，启动时拉取配置：

```typescript
import { defineStore } from 'pinia'
import { get } from '@/api/client'
import { protectVideoUrl } from '@/utils/video'
import { resolveMediaUrl } from '@/utils/media'

interface SiteConfig {
  promo_title: string
  daily_sentence: string
  bg_url?: string
  bg_videos?: string[]
  featured_book_id?: number
}

export const useSiteStore = defineStore('site', {
  state: () => ({
    config: null as SiteConfig | null,
    loaded: false,
  }),
  getters: {
    bgVideos: (s) =>
      (s.config?.bg_videos || []).map((v) => protectVideoUrl(v, 'homePage')),
    promoTitle: (s) => s.config?.promo_title || '宇小兔读书',
    dailySentence: (s) => s.config?.daily_sentence || '',
    bgImage: (s) => resolveMediaUrl(s.config?.bg_url),
  },
  actions: {
    async fetchConfig() {
      const data = await get<SiteConfig>('/site/config', false)
      this.config = data
      this.loaded = true
    },
  },
})
```

### 6.6 User Store `src/stores/user.ts`

```typescript
import { defineStore } from 'pinia'
import { get, post } from '@/api/client'
import { setToken, clearToken } from '@/utils/storage'

interface UserInfo {
  id: number
  username: string
  nickname?: string
  is_vip?: boolean
}

export const useUserStore = defineStore('user', {
  state: () => ({
    user: null as UserInfo | null,
    token: '',
  }),
  getters: {
    isLoggedIn: (s) => !!s.user,
    isVip: (s) => !!s.user?.is_vip,
  },
  actions: {
    async login(username: string, password: string) {
      const data = await post<{ token: string; user: UserInfo }>(
        '/auth/login',
        { username, password },
        false
      )
      setToken(data.token)
      this.user = data.user
    },
    async fetchMe() {
      try {
        this.user = await get<UserInfo>('/auth/me')
      } catch {
        this.user = null
      }
    },
    logout() {
      clearToken()
      this.user = null
    },
  },
})
```

```typescript
import { get, post } from './client'
import type { Book, BookListResult, PlayAuth } from '@/types/book'

export function listBooks(params: Record<string, unknown>) {
  const qs = Object.entries(params)
    .filter(([, v]) => v !== undefined && v !== '')
    .map(([k, v]) => `${k}=${encodeURIComponent(String(v))}`)
    .join('&')
  return get<BookListResult>(`/books?${qs}`, false)
}

export function getBook(id: number) {

}

export function getPlayAuth(id: number) {
  return get<PlayAuth>(`/books/${id}/play-auth`)
}

export function reportPlay(id: number, seconds: number) {
  return post(`/books/${id}/play-report`, { seconds })
}
```

```vue
<template>
  <view class="home">
```

## 十三、与 Web / Admin 协同开发流程

```text
1. Admin 配置内容（书籍、视频、每日、会员套餐、站点氛围）
2. Web 用户端验证展示正确
3. 小程序调用相同 API 验证（无需改 Admin）
4. 若 API 字段变更：
   a. 先改 Python/Java 后端 + Admin
   b. 同步改 Web frontend
   c. 同步改小程序 types + 页面
5. 小程序 UI 迭代独立发版，内容仍由 Admin 驱动
```

**禁止：** 在小程序仓库写死书籍数据或独立维护 JSON 配置。
